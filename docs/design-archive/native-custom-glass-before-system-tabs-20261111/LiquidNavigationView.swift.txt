import Flutter
import UIKit

/// The persistent Flutter navigation host owns one native view. UIKit renders
/// the glass; Flutter remains authoritative for routes, profile, and audio.
final class LiquidNavigationViewFactory: NSObject, FlutterPlatformViewFactory {
  private let messenger: FlutterBinaryMessenger

  init(messenger: FlutterBinaryMessenger) {
    self.messenger = messenger
    super.init()
  }

  func createArgsCodec() -> FlutterMessageCodec & NSObjectProtocol {
    FlutterStandardMessageCodec.sharedInstance()
  }

  func create(
    withFrame frame: CGRect,
    viewIdentifier viewId: Int64,
    arguments args: Any?
  ) -> FlutterPlatformView {
    LiquidNavigationPlatformView(
      frame: frame,
      identifier: viewId,
      arguments: args,
      messenger: messenger
    )
  }
}

private final class LiquidNavigationPlatformView: NSObject, FlutterPlatformView {
  private let navigationView: LiquidNavigationView
  private let channel: FlutterMethodChannel

  init(
    frame: CGRect,
    identifier: Int64,
    arguments: Any?,
    messenger: FlutterBinaryMessenger
  ) {
    navigationView = LiquidNavigationView(frame: frame)
    channel = FlutterMethodChannel(
      name: "kingdom_kids/liquid_navigation/\(identifier)",
      binaryMessenger: messenger
    )
    super.init()
    if let values = arguments as? [String: Any] {
      navigationView.update(values, animated: false)
    }
    navigationView.onSelect = { [weak self] route in
      self?.channel.invokeMethod("select", arguments: ["route": route])
    }
    channel.setMethodCallHandler { [weak self] call, result in
      guard let self = self else {
        result(nil)
        return
      }
      guard call.method == "update" else {
        result(FlutterMethodNotImplemented)
        return
      }
      guard let values = call.arguments as? [String: Any],
            let route = values["selectedRoute"] as? String,
            LiquidNavigationView.routes.contains(route) else {
        result(FlutterError(
          code: "invalid_navigation_update",
          message: "A supported selected route is required.",
          details: nil
        ))
        return
      }
      self.navigationView.update(values, animated: true)
      result(nil)
    }
  }

  func view() -> UIView { navigationView }

  deinit {
    channel.setMethodCallHandler(nil)
  }
}

private final class LiquidNavigationView: UIView {
  static let routes = ["/", "/stories", "/games", "/profile"]
  private static let titles = ["Home", "Stories", "Games"]
  private static let symbols = ["house.fill", "book.closed.fill", "puzzlepiece.extension.fill"]
  private static let selectedInk = UIColor(red: 20 / 255, green: 87 / 255, blue: 1, alpha: 1)

  var onSelect: ((String) -> Void)?

  private let dockGlass = LiquidNavigationView.makeGlass()
  private let selectionGlass = LiquidNavigationView.makeGlass(selection: true)
  private let profileGlass = LiquidNavigationView.makeGlass()
  private let profileButton = UIButton(type: .custom)
  private let profileImageView = UIImageView()
  private var buttons: [LiquidNavigationTabButton] = []
  private var selectedRoute = "/"
  private var reduceMotion = false
  private var lastSize = CGSize.zero
  private var accessibilityObserver: NSObjectProtocol?

  override init(frame: CGRect) {
    super.init(frame: frame)
    backgroundColor = .clear
    isOpaque = false
    overrideUserInterfaceStyle = .light
    isAccessibilityElement = false
    addSubview(dockGlass)
    // The lens is a single moving native glass view, not one fading tile per tab.
    dockGlass.contentView.addSubview(selectionGlass)
    selectionGlass.isUserInteractionEnabled = false
    for index in 0..<3 {
      let button = LiquidNavigationTabButton(
        title: Self.titles[index],
        symbol: Self.symbols[index]
      )
      button.tag = index
      button.addTarget(self, action: #selector(tabTapped(_:)), for: .touchUpInside)
      dockGlass.contentView.addSubview(button)
      buttons.append(button)
    }

    addSubview(profileGlass)
    profileGlass.contentView.addSubview(profileButton)
    profileButton.accessibilityLabel = "Profile"
    profileButton.accessibilityIdentifier = "liquid-navigation-profile"
    profileButton.addTarget(self, action: #selector(profileTapped), for: .touchUpInside)
    profileImageView.contentMode = .scaleAspectFill
    profileImageView.clipsToBounds = true
    profileImageView.isUserInteractionEnabled = false
    profileButton.addSubview(profileImageView)
    clearProfileImage()
    accessibilityElements = buttons + [profileButton]
    accessibilityObserver = NotificationCenter.default.addObserver(
      forName: UIAccessibility.reduceMotionStatusDidChangeNotification,
      object: nil,
      queue: .main
    ) { [weak self] _ in
      guard let self = self, self.motionIsReduced else { return }
      self.selectionGlass.layer.removeAllAnimations()
      self.applySelection(animated: false)
    }
    applySelection(animated: false)
  }

  required init?(coder: NSCoder) { fatalError("init(coder:) has not been implemented") }

  private var motionIsReduced: Bool {
    reduceMotion || UIAccessibility.isReduceMotionEnabled
  }

  func update(_ values: [String: Any], animated: Bool) {
    guard let route = values["selectedRoute"] as? String,
          Self.routes.contains(route) else { return }
    let selectionChanged = route != selectedRoute
    selectedRoute = route
    reduceMotion = values["reduceMotion"] as? Bool ?? false
    if let imageBytes = values["profileImage"] as? FlutterStandardTypedData,
       let image = UIImage(data: imageBytes.data) {
      profileImageView.image = image
      profileImageView.tintColor = nil
    } else {
      clearProfileImage()
    }
    // The native tap begins its glide before Dart replaces the page. Its route
    // acknowledgement (and later avatar loads) must not cancel that in-flight
    // animation merely because it confirms the same selected route.
    if selectionChanged || !animated || motionIsReduced {
      applySelection(animated: animated && selectionChanged)
    }
  }

  override func layoutSubviews() {
    super.layoutSubviews()
    let height = min(bounds.height, 69)
    let profileWidth = min(height, bounds.width)
    let gap: CGFloat = 10
    let dockWidth = max(0, bounds.width - profileWidth - gap)
    dockGlass.frame = CGRect(x: 0, y: 0, width: dockWidth, height: height)
    profileGlass.frame = CGRect(x: dockWidth + gap, y: 0, width: profileWidth, height: height)
    Self.makeCapsule(dockGlass)
    Self.makeCapsule(profileGlass)
    profileButton.frame = profileGlass.bounds
    let avatarSize = min(52, max(0, height - 12))
    profileImageView.frame = CGRect(
      x: (profileWidth - avatarSize) / 2,
      y: (height - avatarSize) / 2,
      width: avatarSize,
      height: avatarSize
    )
    profileImageView.layer.cornerRadius = avatarSize / 2
    let inset: CGFloat = 5
    let itemWidth = max(0, (dockWidth - inset * 2) / 3)
    for (index, button) in buttons.enumerated() {
      button.frame = CGRect(x: inset + CGFloat(index) * itemWidth, y: inset,
                            width: itemWidth, height: max(0, height - inset * 2))
    }
    if lastSize != bounds.size {
      selectionGlass.layer.removeAllAnimations()
      lastSize = bounds.size
      applySelection(animated: false)
    }
    Self.makeCapsule(selectionGlass)
  }

  override func point(inside point: CGPoint, with event: UIEvent?) -> Bool {
    // A transparent space between the two glass pieces does not block content.
    dockGlass.frame.contains(point) || profileGlass.frame.contains(point)
  }

  private func applySelection(animated: Bool) {
    let index = Self.routes.firstIndex(of: selectedRoute) ?? 0
    for (buttonIndex, button) in buttons.enumerated() {
      button.updateSelection(buttonIndex == index, ink: Self.selectedInk)
    }
    let profileSelected = index == 3
    profileButton.accessibilityTraits = profileSelected ? [.button, .selected] : [.button]
    profileGlass.contentView.layer.borderWidth = profileSelected ? 1.5 : 0
    profileGlass.contentView.layer.borderColor = Self.selectedInk.withAlphaComponent(0.5).cgColor
    profileGlass.contentView.layer.cornerRadius = min(profileGlass.bounds.width, profileGlass.bounds.height) / 2

    // Profile stays outside the shared dock: selecting it withdraws the lens
    // rather than pretending it is a fourth tab inside the capsule.
    let targetFrame = buttons[min(index, 2)].frame
    let changes = {
      self.selectionGlass.frame = targetFrame
      self.selectionGlass.alpha = profileSelected ? 0 : 1
      Self.makeCapsule(self.selectionGlass)
    }
    if !animated || motionIsReduced || dockGlass.bounds.width == 0 {
      selectionGlass.layer.removeAllAnimations()
      UIView.performWithoutAnimation(changes)
    } else {
      UIView.animate(
        withDuration: 0.42,
        delay: 0,
        usingSpringWithDamping: 0.86,
        initialSpringVelocity: 0,
        options: [.beginFromCurrentState, .allowUserInteraction],
        animations: changes
      )
    }
  }

  @objc private func tabTapped(_ sender: UIButton) {
    select(Self.routes[sender.tag])
  }

  @objc private func profileTapped() { select("/profile") }

  private func select(_ route: String) {
    guard Self.routes.contains(route), route != selectedRoute else { return }
    selectedRoute = route
    applySelection(animated: true)
    onSelect?(route)
  }

  private func clearProfileImage() {
    profileImageView.image = UIImage(systemName: "person.crop.circle.fill")
    profileImageView.tintColor = .label
  }

  private static func makeGlass(selection: Bool = false) -> UIVisualEffectView {
    let effect: UIVisualEffect
    if #available(iOS 26.0, *) {
      let glass = UIGlassEffect(style: .clear)
      glass.isInteractive = true
      // The selected lens keeps actual clear system glass, with only a slight
      // blue tint. No white fill or synthetic frosted gradient covers content.
      if selection { glass.tintColor = selectedInk.withAlphaComponent(0.07) }
      effect = glass
    } else {
      effect = UIBlurEffect(style: .systemUltraThinMaterialLight)
    }
    let view = UIVisualEffectView(effect: effect)
    view.backgroundColor = .clear
    view.isOpaque = false
    view.clipsToBounds = true
    return view
  }

  private static func makeCapsule(_ view: UIView) {
    if #available(iOS 26.0, *) {
      view.cornerConfiguration = .capsule()
    } else {
      view.layer.cornerRadius = min(view.bounds.width, view.bounds.height) / 2
      view.layer.cornerCurve = .continuous
    }
  }

  deinit {
    if let accessibilityObserver = accessibilityObserver {
      NotificationCenter.default.removeObserver(accessibilityObserver)
    }
  }
}

private final class LiquidNavigationTabButton: UIButton {
  private let symbolView: UIImageView
  private let titleView = UILabel()

  init(title: String, symbol: String) {
    symbolView = UIImageView(image: UIImage(systemName: symbol))
    super.init(frame: .zero)
    accessibilityLabel = title
    accessibilityIdentifier = "liquid-navigation-\(title.lowercased())"
    isAccessibilityElement = true
    symbolView.contentMode = .scaleAspectFit
    symbolView.isUserInteractionEnabled = false
    titleView.text = title
    titleView.font = .systemFont(ofSize: 10.5, weight: .semibold)
    titleView.textAlignment = .center
    titleView.adjustsFontForContentSizeCategory = true
    titleView.isUserInteractionEnabled = false
    addSubview(symbolView)
    addSubview(titleView)
  }

  required init?(coder: NSCoder) { fatalError("init(coder:) has not been implemented") }

  override func layoutSubviews() {
    super.layoutSubviews()
    symbolView.frame = CGRect(x: (bounds.width - 24) / 2, y: 9, width: 24, height: 24)
    titleView.frame = CGRect(x: 2, y: 35, width: max(0, bounds.width - 4), height: 15)
  }

  func updateSelection(_ selected: Bool, ink: UIColor) {
    let color: UIColor = selected ? ink : .label
    symbolView.tintColor = color
    titleView.textColor = color
    accessibilityTraits = selected ? [.button, .selected] : [.button]
  }
}
