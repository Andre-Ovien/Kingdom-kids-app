import 'package:flutter/material.dart';

enum AuthScreenMode { register, login }

/// Parent-only authentication. A child profile is created in the next step and
/// never asks a child for an email address or password.
class AuthScreen extends StatefulWidget {
  const AuthScreen({
    super.key,
    this.initialMode = AuthScreenMode.register,
    this.repository,
    this.externalUrlLauncher,
  });

  final AuthScreenMode initialMode;
  final AuthRepository? repository;
  final Future<bool> Function(Uri uri)? externalUrlLauncher;

  @override
  State<AuthScreen> createState() => _AuthScreenState();
}

class _AuthScreenState extends State<AuthScreen>
    with SingleTickerProviderStateMixin {
  var _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  final _confirmPasswordController = TextEditingController();
  final _passwordFocusNode = FocusNode();
  final _confirmPasswordFocusNode = FocusNode();

  late final AuthRepository _repository;
  late final AnimationController _modeController;
  late AuthScreenMode _mode;
  bool _isSubmitting = false;
  bool _showValidationErrors = false;
  bool _showPassword = false;
  bool _showConfirmPassword = false;
  bool _reduceMotion = false;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _repository = widget.repository ?? AuthRepository();
    _mode = widget.initialMode;
    _modeController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 680),
      value: _mode == AuthScreenMode.register ? 1 : 0,
    );
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    _reduceMotion = MediaQuery.of(context).disableAnimations;
    if (_reduceMotion) {
      _modeController.value = _mode == AuthScreenMode.register ? 1 : 0;
    }
  }

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    _confirmPasswordController.dispose();
    _passwordFocusNode.dispose();
    _confirmPasswordFocusNode.dispose();
    _modeController.dispose();
    super.dispose();
  }

  void _changeMode(AuthScreenMode mode) {
    if (mode == _mode || _isSubmitting) return;
    FocusManager.instance.primaryFocus?.unfocus();
    setState(() {
      _mode = mode;
      _errorMessage = null;
      _showValidationErrors = false;
      _formKey = GlobalKey<FormState>();
    });
    final target = mode == AuthScreenMode.register ? 1.0 : 0.0;
    if (_reduceMotion) {
      _modeController.value = target;
    } else {
      _modeController.animateTo(
        target,
        duration: Duration(
          milliseconds: (680 * (target - _modeController.value).abs()).round(),
        ),
        curve: Curves.easeInOutCubic,
      );
    }
  }

  Future<void> _submit() async {
    if (_isSubmitting || _modeController.isAnimating) {
      return;
    }

    FocusManager.instance.primaryFocus?.unfocus();
    if (!(_formKey.currentState?.validate() ?? false)) {
      setState(() => _showValidationErrors = true);
      return;
    }

    setState(() {
      _isSubmitting = true;
      _showValidationErrors = false;
      _errorMessage = null;
    });

    try {
      if (_mode == AuthScreenMode.register) {
        final pending = await _repository.register(
          RegisterParentRequest(
            email: _emailController.text,
            password: _passwordController.text,
          ),
        );
        if (mounted) {
          _openVerification(pending);
        }
        return;
      }

      final session = await _repository.login(
        LoginRequest(
          email: _emailController.text,
          password: _passwordController.text,
        ),
      );

      if (!mounted) {
        return;
      }
      _continueToProfiles(session);
    } on AppException catch (error) {
      if (mounted &&
          error.code == 'email_verification_required' &&
          _openVerificationFromError(error)) {
        return;
      }
      if (mounted) {
        setState(() => _errorMessage = error.userMessage);
      }
    } catch (_) {
      if (mounted) {
        setState(
          () => _errorMessage = 'Something went wrong. Please try again.',
        );
      }
    } finally {
      if (mounted) {
        setState(() => _isSubmitting = false);
      }
    }
  }

  void _openVerification(PendingEmailVerification pending) {
    Navigator.of(
      context,
    ).pushNamed(AppRoutes.emailVerification, arguments: pending);
  }

  bool _openVerificationFromError(AppException error) {
    final pending = PendingEmailVerification.fromJson(
      error.details,
      fallbackEmail: _emailController.text.trim().toLowerCase(),
    );
    if (pending.challengeId.isEmpty ||
        (pending.destinationHint.isEmpty && pending.email.isEmpty)) {
      return false;
    }
    _openVerification(pending);
    return true;
  }

  void _continueToProfiles(AuthSession session) {
    Navigator.of(context).pushNamedAndRemoveUntil(
      session.hasProfiles ? AppRoutes.profilePicker : AppRoutes.kidProfileSetup,
      (_) => false,
      arguments: session.profiles,
    );
  }

  Future<void> _openPrivacyPolicy() async {
    final uri = Uri.parse(AppConstants.privacyPolicyUrl);
    try {
      final opened =
          await (widget.externalUrlLauncher?.call(uri) ??
              launchUrl(uri, mode: LaunchMode.externalApplication));
      if (!opened && mounted) {
        _showPrivacyError();
      }
    } catch (_) {
      if (mounted) _showPrivacyError();
    }
  }

  void _showPrivacyError() {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        const SnackBar(
          content: Text('We could not open the privacy policy on this device.'),
        ),
      );
  }

  @override
  Widget build(BuildContext context) {
    final isDark = Theme.of(context).brightness == Brightness.dark;
    final palette = _AuthPalette.forMode(_mode, isDark: isDark);
    final media = MediaQuery.of(context);
    final keyboardOpen = media.viewInsets.bottom > 0;
    final textScale = media.textScaler.scale(14) / 14;
    return Scaffold(
      backgroundColor: palette.background,
      body: GestureDetector(
        behavior: HitTestBehavior.translucent,
        onTap: () => FocusManager.instance.primaryFocus?.unfocus(),
        child: AnimatedBuilder(
          animation: _modeController,
          builder: (context, _) => LayoutBuilder(
            builder: (context, constraints) {
              final progress = _modeController.value;
              final compact =
                  keyboardOpen ||
                  constraints.maxHeight - media.padding.vertical < 650 ||
                  textScale > 1.3 ||
                  _showValidationErrors ||
                  _errorMessage != null;
              final bottomInset = keyboardOpen ? 0.0 : media.padding.bottom;
              final headerHeight = compact
                  ? media.padding.top +
                        math.max(48.0, media.textScaler.scale(26) * 1.2 + 16)
                  : math.min(328.0, constraints.maxHeight * 0.39) *
                            (1 - progress) +
                        math.min(210.0, constraints.maxHeight * 0.25) *
                            progress;
              final content = _buildForm(
                palette,
                progress,
                compact,
                keyboardOpen,
              );
              return Stack(
                fit: StackFit.expand,
                children: [
                  Positioned(
                    top: 0,
                    left: 0,
                    right: 0,
                    height: headerHeight,
                    child: AuthWaveHeader(
                      progress: progress,
                      height: headerHeight,
                      topInset: media.padding.top,
                      compact: compact,
                      color: palette.action,
                    ),
                  ),
                  Positioned(
                    top: headerHeight,
                    left: 0,
                    right: 0,
                    bottom: bottomInset,
                    child: Transform.translate(
                      offset: Offset(
                        0,
                        _reduceMotion ? 0 : 8 * math.sin(progress * math.pi),
                      ),
                      child: textScale >= 1.7
                          ? SingleChildScrollView(
                              padding: const EdgeInsets.fromLTRB(
                                24,
                                12,
                                24,
                                12,
                              ),
                              keyboardDismissBehavior:
                                  ScrollViewKeyboardDismissBehavior.onDrag,
                              child: Center(
                                child: ConstrainedBox(
                                  constraints: const BoxConstraints(
                                    maxWidth: 430,
                                  ),
                                  child: content,
                                ),
                              ),
                            )
                          : Padding(
                              key: const Key('auth-register-viewport'),
                              padding: EdgeInsets.fromLTRB(
                                24,
                                compact ? 4 : 32,
                                24,
                                compact ? 6 : 8,
                              ),
                              child: Align(
                                alignment: Alignment.topCenter,
                                child: ConstrainedBox(
                                  constraints: const BoxConstraints(
                                    maxWidth: 430,
                                  ),
                                  child: content,
                                ),
                              ),
                            ),
                    ),
                  ),
                ],
              );
            },
          ),
        ),
      ),
    );
  }

  Widget _buildForm(
    _AuthPalette palette,
    double progress,
    bool compact,
    bool keyboardOpen,
  ) {
    final gap = compact ? 4.0 : 10.0;
    final registering = _mode == AuthScreenMode.register;
    return Column(
      mainAxisSize: MainAxisSize.min,
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        if (!keyboardOpen) ...[
          _ModeCaption(
            progress: progress,
            compact: compact,
            color: palette.foreground,
          ),
          SizedBox(height: compact ? 10 : 28),
        ],
        AutofillGroup(
          child: Form(
            key: _formKey,
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                _AuthTextField(
                  key: const Key('auth-email-field'),
                  controller: _emailController,
                  palette: palette,
                  compact: true,
                  label: 'Email address',
                  hint: 'you@example.com',
                  keyboardType: TextInputType.emailAddress,
                  autofillHints: const [AutofillHints.email],
                  validator: _validateEmail,
                ),
                SizedBox(height: gap),
                _AuthTextField(
                  key: const Key('auth-password-field'),
                  controller: _passwordController,
                  focusNode: _passwordFocusNode,
                  palette: palette,
                  compact: true,
                  label: 'Password',
                  hint: registering ? 'At least 8 characters' : 'Your password',
                  obscureText: !_showPassword,
                  autofillHints: [
                    registering
                        ? AutofillHints.newPassword
                        : AutofillHints.password,
                  ],
                  suffixIcon: IconButton(
                    tooltip: _showPassword ? 'Hide password' : 'Show password',
                    onPressed: () =>
                        setState(() => _showPassword = !_showPassword),
                    icon: Icon(
                      _showPassword
                          ? Icons.visibility_off_rounded
                          : Icons.visibility_rounded,
                    ),
                  ),
                  validator: _validatePassword,
                  onFieldSubmitted: (_) {
                    if (!registering) _submit();
                  },
                ),
                if (progress < 1)
                  ClipRect(
                    child: Align(
                      alignment: Alignment.topCenter,
                      heightFactor: 1 - progress,
                      child: Opacity(
                        opacity: 1 - progress,
                        child: IgnorePointer(
                          ignoring: registering,
                          child: ExcludeSemantics(
                            excluding: registering,
                            child: Align(
                              alignment: Alignment.centerRight,
                              child: TextButton(
                                key: const Key('auth-forgot-password'),
                                onPressed: _isSubmitting
                                    ? null
                                    : () => Navigator.of(
                                        context,
                                      ).pushNamed(AppRoutes.passwordReset),
                                style: TextButton.styleFrom(
                                  foregroundColor: palette.action,
                                  minimumSize: const Size(48, 44),
                                ),
                                child: const Text(
                                  'Forgot password?',
                                  style: TextStyle(
                                    fontFamily: 'OnboardingPoppins',
                                    fontSize: 12,
                                    fontWeight: FontWeight.w600,
                                  ),
                                ),
                              ),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
                if (progress > 0)
                  ClipRect(
                    child: Align(
                      alignment: Alignment.topCenter,
                      heightFactor: progress,
                      child: Opacity(
                        opacity: progress,
                        child: IgnorePointer(
                          ignoring: !registering,
                          child: ExcludeSemantics(
                            excluding: !registering,
                            child: Padding(
                              padding: EdgeInsets.only(top: gap),
                              child: _AuthTextField(
                                key: const Key('auth-confirm-password-field'),
                                controller: _confirmPasswordController,
                                focusNode: _confirmPasswordFocusNode,
                                palette: palette,
                                compact: true,
                                label: 'Confirm password',
                                hint: 'One more time',
                                obscureText: !_showConfirmPassword,
                                autofillHints: const [
                                  AutofillHints.newPassword,
                                ],
                                suffixIcon: IconButton(
                                  tooltip: _showConfirmPassword
                                      ? 'Hide password'
                                      : 'Show password',
                                  onPressed: () => setState(
                                    () => _showConfirmPassword =
                                        !_showConfirmPassword,
                                  ),
                                  icon: Icon(
                                    _showConfirmPassword
                                        ? Icons.visibility_off_rounded
                                        : Icons.visibility_rounded,
                                  ),
                                ),
                                validator: (value) =>
                                    registering &&
                                        value != _passwordController.text
                                    ? 'Those passwords do not match.'
                                    : null,
                                onFieldSubmitted: (_) => _submit(),
                              ),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
              ],
            ),
          ),
        ),
        if (_errorMessage != null) ...[
          const SizedBox(height: 8),
          _ErrorNotice(message: _errorMessage!, palette: palette),
        ],
        SizedBox(height: compact ? 8 : 14),
        FilledButton(
          key: const Key('auth-submit-button'),
          style: FilledButton.styleFrom(
            backgroundColor: palette.action,
            foregroundColor: palette.onAction,
            disabledBackgroundColor: palette.action.withValues(alpha: 0.48),
            disabledForegroundColor: palette.onAction.withValues(alpha: 0.7),
            minimumSize: const Size.fromHeight(52),
            shape: RoundedRectangleBorder(
              borderRadius: BorderRadius.circular(20),
            ),
            textStyle: const TextStyle(
              fontFamily: 'OnboardingPoppins',
              fontSize: 15,
              fontWeight: FontWeight.w600,
            ),
          ),
          onPressed: _isSubmitting || _modeController.isAnimating
              ? null
              : _submit,
          child: _isSubmitting
              ? SizedBox.square(
                  dimension: 22,
                  child: CircularProgressIndicator(
                    strokeWidth: 2.4,
                    color: palette.onAction,
                  ),
                )
              : Text(registering ? 'Create account' : 'Sign in'),
        ),
        _AuthPrivacyPolicyLink(
          mode: _mode,
          palette: palette,
          onPressed: _isSubmitting ? null : _openPrivacyPolicy,
        ),
        _InlineModeLink(mode: _mode, palette: palette, onChanged: _changeMode),
        const SizedBox(height: 4),
        _ParentSafetyNote(palette: palette, compact: true),
      ],
    );
  }

  String? _validateEmail(String? value) {
    final email = value?.trim() ?? '';
    if (email.isEmpty) {
      return 'Please enter your email address.';
    }
    if (!RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(email)) {
      return 'Enter a valid email address.';
    }
    return null;
  }

  String? _validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Please enter your password.';
    }
    if (_mode == AuthScreenMode.register && value.length < 8) {
      return 'Use at least 8 characters.';
    }
    return null;
  }
}

class _AuthPalette {
  const _AuthPalette({
    required this.background,
    required this.foreground,
    required this.muted,
    required this.fieldFill,
    required this.fieldText,
    required this.fieldHint,
    required this.focus,
    required this.action,
    required this.onAction,
    required this.errorBackground,
    required this.errorText,
  });
  factory _AuthPalette.forMode(AuthScreenMode mode, {required bool isDark}) =>
      isDark
      ? const _AuthPalette(
          background: Color(0xFF111C35),
          foreground: Color(0xFFF2F6FF),
          muted: Color(0xFFB0BED7),
          fieldFill: Color(0xFF1C2D4C),
          fieldText: Color(0xFFF2F6FF),
          fieldHint: Color(0xFFA9BAD8),
          focus: Color(0xFF84ADFF),
          action: Color(0xFF1457FF),
          onAction: Colors.white,
          errorBackground: Color(0xFF4C2738),
          errorText: Color(0xFFFFC8D4),
        )
      : const _AuthPalette(
          background: Color(0xFFF3F6FD),
          foreground: Color(0xFF23375B),
          muted: Color(0xFF6C7C99),
          fieldFill: Colors.white,
          fieldText: Color(0xFF23375B),
          fieldHint: Color(0xFF6C7C99),
          focus: Color(0xFF1457FF),
          action: Color(0xFF1457FF),
          onAction: Colors.white,
          errorBackground: Color(0xFFFFEBEF),
          errorText: Color(0xFFA3334E),
        );
  final Color background,
      foreground,
      muted,
      fieldFill,
      fieldText,
      fieldHint,
      focus,
      action,
      onAction,
      errorBackground,
      errorText;
}

class _ModeCaption extends StatelessWidget {
  const _ModeCaption({
    required this.progress,
    required this.compact,
    required this.color,
  });
  final double progress;
  final bool compact;
  final Color color;
  @override
  Widget build(BuildContext context) {
    final style = TextStyle(
      fontFamily: 'OnboardingPoppins',
      fontSize: compact ? 21 : 25,
      height: 1.2,
      fontWeight: FontWeight.w600,
      color: color,
      letterSpacing: -0.5,
    );
    return SizedBox(
      height: MediaQuery.textScalerOf(context).scale(compact ? 26 : 31),
      child: Stack(
        alignment: Alignment.center,
        children: [
          if (progress < 1)
            Opacity(
              opacity: (1 - 2 * progress).clamp(0.0, 1.0),
              child: Transform.translate(
                offset: Offset(0, -8 * progress),
                child: Text('Welcome back!', style: style),
              ),
            ),
          if (progress > 0)
            Opacity(
              opacity: (2 * progress - 1).clamp(0.0, 1.0),
              child: Transform.translate(
                offset: Offset(0, 8 * (1 - progress)),
                child: Text('Hey, welcome!', style: style),
              ),
            ),
        ],
      ),
    );
  }
}

class _InlineModeLink extends StatelessWidget {

    required this.mode,

    required this.onChanged,
  });

    final prefix = isRegister
        ? 'Already have an account? '

    final action = isRegister ? 'Sign in' : 'Sign up';

      key: const Key('auth-mode-link'),

      ),
      child: Text.rich(
        TextSpan(
          children: [
            TextSpan(
              text: prefix,
              style: TextStyle(
                color: palette.muted,
                fontWeight: FontWeight.w500,
              ),
            ),
            TextSpan(
              text: action,
              style: TextStyle(
                color: palette.foreground,
                fontWeight: FontWeight.w900,
                decoration: TextDecoration.underline,
                decorationColor: palette.foreground,

            ),
          ],
        ),
        textAlign: TextAlign.center,
        style: const TextStyle(
          fontFamily: 'OnboardingPoppins',
          fontSize: 12.5,
          height: 1.3,
        ),
      ),
    );
  }
}

class _AuthTextField extends StatelessWidget {
  const _AuthTextField({
    super.key,
    required this.controller,
    this.focusNode,
    this.compact = false,
    required this.palette,
    required this.label,
    required this.hint,
    this.keyboardType,
    this.autofillHints,
    this.obscureText = false,
    this.suffixIcon,
    this.validator,
    this.onFieldSubmitted,
  });

  final TextEditingController controller;
  final FocusNode? focusNode;
  final bool compact;
  final _AuthPalette palette;
  final String label;
  final String hint;
  final TextInputType? keyboardType;
  final Iterable<String>? autofillHints;
  final bool obscureText;
  final Widget? suffixIcon;
  final FormFieldValidator<String>? validator;
  final ValueChanged<String>? onFieldSubmitted;

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: controller,
      focusNode: focusNode,
      keyboardType: keyboardType,
      autofillHints: autofillHints,
      obscureText: obscureText,
      enableSuggestions: !obscureText,
      autocorrect: !obscureText,
      textInputAction: onFieldSubmitted == null
          ? TextInputAction.next
          : TextInputAction.done,
      validator: validator,
      onFieldSubmitted: onFieldSubmitted,
      cursorColor: palette.focus,
      style: TextStyle(
        fontFamily: 'OnboardingPoppins',
        color: palette.fieldText,
        fontSize: 15,
        fontWeight: FontWeight.w600,
      ),
      decoration: InputDecoration(
        labelText: label,
        floatingLabelBehavior: FloatingLabelBehavior.always,
        suffixIcon: suffixIcon,
        suffixIconColor: palette.fieldHint,
        filled: false,
        labelStyle: TextStyle(
          fontFamily: 'OnboardingPoppins',
          fontSize: 14,
          height: 1.2,
          color: palette.fieldHint,
          fontWeight: FontWeight.w600,
        ),
        floatingLabelStyle: TextStyle(
          fontFamily: 'OnboardingPoppins',
          fontSize: 14,
          height: 1.2,
          color: palette.focus,
          fontWeight: FontWeight.w600,
        ),
        hintStyle: TextStyle(
          fontFamily: 'OnboardingPoppins',
          color: palette.fieldHint,
          fontWeight: FontWeight.w500,
        ),
        errorStyle: TextStyle(
          fontFamily: 'OnboardingPoppins',
          color: palette.errorText,
          fontWeight: FontWeight.w700,
        ),
        isDense: compact,
        contentPadding: EdgeInsets.symmetric(
          horizontal: 4,
          vertical: compact ? 8 : 12,
        ),
        border: _border(),
        enabledBorder: _border(),
        focusedBorder: _border(color: palette.focus, width: 2),
        errorBorder: _border(color: palette.errorText, width: 1.5),
        focusedErrorBorder: _border(color: palette.errorText, width: 2),
      ),
    );
  }

  UnderlineInputBorder _border({
    Color color = const Color(0xFFD8E2F4),
    double width = 1.2,
  }) {
    return UnderlineInputBorder(
      borderSide: BorderSide(color: color, width: width),
    );
  }
}

class _ErrorNotice extends StatelessWidget {
  const _ErrorNotice({required this.message, required this.palette});

  final String message;
  final _AuthPalette palette;

  @override
  Widget build(BuildContext context) {
    return Semantics(
      liveRegion: true,
      child: Container(
        padding: const EdgeInsets.all(14),
        decoration: BoxDecoration(
          color: palette.errorBackground,
          borderRadius: BorderRadius.circular(18),
        ),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Icon(
              Icons.info_outline_rounded,
