enum AuthScreenMode { register, login }

/// Parent-only authentication. A child profile is created in the next step and
/// never asks a child for an email address or password.
class AuthScreen extends StatefulWidget {
  const AuthScreen({
    super.key,
    this.initialMode = AuthScreenMode.register,
    this.repository,
    this.externalUrlLauncher,
  });

  final AuthScreenMode initialMode;
  final AuthRepository? repository;
  final Future<bool> Function(Uri uri)? externalUrlLauncher;

  @override
  State<AuthScreen> createState() => _AuthScreenState();
}

class _AuthScreenState extends State<AuthScreen>
    with SingleTickerProviderStateMixin {
  final _formKey = GlobalKey<FormState>();
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  final _confirmPasswordController = TextEditingController();
  final _passwordFocusNode = FocusNode();
  final _confirmPasswordFocusNode = FocusNode();

  late final AuthRepository _repository;
  late final AnimationController _faceController;
  late final Listenable _secureFieldFocus;
  late AuthScreenMode _mode;
  bool _isSubmitting = false;
  bool _showValidationErrors = false;
  bool _showPassword = false;
  bool _showConfirmPassword = false;
  bool _reduceMotion = false;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _repository = widget.repository ?? AuthRepository();
    _mode = widget.initialMode;
    _faceController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 2200),
    );
    _secureFieldFocus = Listenable.merge([
      _passwordFocusNode,
      _confirmPasswordFocusNode,
    ]);
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    final reduceMotion = MediaQuery.of(context).disableAnimations;
    if (reduceMotion == _reduceMotion && _faceController.isAnimating) {
      return;
    }

    _reduceMotion = reduceMotion;
    if (_reduceMotion) {
      _faceController
        ..stop()
        ..value = 0.5;
    } else if (!_faceController.isAnimating) {
      _faceController.repeat(reverse: true);
    }
  }

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    _confirmPasswordController.dispose();
    _passwordFocusNode.dispose();
    _confirmPasswordFocusNode.dispose();
    _faceController.dispose();
    super.dispose();
  }

  bool get _isEnteringPassword =>
      _passwordFocusNode.hasFocus ||
      (_mode == AuthScreenMode.register && _confirmPasswordFocusNode.hasFocus);

  void _changeMode(AuthScreenMode mode) {
    if (mode == _mode) {
      return;
    }
    AppAudioScope.maybeOf(context)?.playEffect(AppSoundCue.select);
    if (mode == AuthScreenMode.login) {
      _confirmPasswordFocusNode.unfocus();
    }
    setState(() {
      _mode = mode;
      _errorMessage = null;
      _showValidationErrors = false;
    });
  }

  Future<void> _submit() async {
    if (_isSubmitting) {
      return;
    }

    FocusManager.instance.primaryFocus?.unfocus();
    if (!(_formKey.currentState?.validate() ?? false)) {
      setState(() => _showValidationErrors = true);
      return;
    }

    AppAudioScope.maybeOf(context)?.playEffect(AppSoundCue.tap);
    setState(() {
      _isSubmitting = true;
      _showValidationErrors = false;
      _errorMessage = null;
    });

    try {
      if (_mode == AuthScreenMode.register) {
        final pending = await _repository.register(
          RegisterParentRequest(
            email: _emailController.text,
            password: _passwordController.text,
          ),
        );
        if (mounted) {
          _openVerification(pending);
        }
        return;
      }

      final session = await _repository.login(
        LoginRequest(
          email: _emailController.text,
          password: _passwordController.text,
        ),
      );

      if (!mounted) {
        return;
      }
      _continueToProfiles(session);
    } on AppException catch (error) {
      if (mounted &&
          error.code == 'email_verification_required' &&
          _openVerificationFromError(error)) {
        return;
      }
      if (mounted) {
        setState(() => _errorMessage = error.userMessage);
      }
    } catch (_) {
      if (mounted) {
        setState(
          () => _errorMessage = 'Something went wrong. Please try again.',
        );
      }
    } finally {
      if (mounted) {
        setState(() => _isSubmitting = false);
      }
    }
  }

  void _openVerification(PendingEmailVerification pending) {
    Navigator.of(
      context,
    ).pushNamed(AppRoutes.emailVerification, arguments: pending);
  }

  bool _openVerificationFromError(AppException error) {
    final pending = PendingEmailVerification.fromJson(
      error.details,
      fallbackEmail: _emailController.text.trim().toLowerCase(),
    );
    if (pending.challengeId.isEmpty ||
        (pending.destinationHint.isEmpty && pending.email.isEmpty)) {
      return false;
    }
    _openVerification(pending);
    return true;
  }

  void _continueToProfiles(AuthSession session) {
    Navigator.of(context).pushNamedAndRemoveUntil(
      session.hasProfiles ? AppRoutes.profilePicker : AppRoutes.kidProfileSetup,
      (_) => false,
      arguments: session.profiles,
    );
  }

  Future<void> _openPrivacyPolicy() async {
    final uri = Uri.parse(AppConstants.privacyPolicyUrl);
    try {
      final opened =
          await (widget.externalUrlLauncher?.call(uri) ??
              launchUrl(uri, mode: LaunchMode.externalApplication));
      if (!opened && mounted) {
        _showPrivacyError();
      }
    } catch (_) {
      if (mounted) _showPrivacyError();
    }
  }

  void _showPrivacyError() {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        const SnackBar(
          content: Text('We could not open the privacy policy on this device.'),
        ),
      );
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final isDark = theme.brightness == Brightness.dark;
    final palette = _AuthPalette.forMode(_mode, isDark: isDark);
    // Scaffold removes the applied keyboard inset from its body MediaQuery.
    final keyboardOpen = MediaQuery.viewInsetsOf(context).bottom > 0;

    return Scaffold(
      backgroundColor: palette.background,
      body: GestureDetector(
        behavior: HitTestBehavior.translucent,
        onTap: () => FocusManager.instance.primaryFocus?.unfocus(),
        child: Stack(
          children: [
            _AuthBackdrop(palette: palette),
            SafeArea(
              child: LayoutBuilder(
                builder: (context, constraints) {
                  final isRegister = _mode == AuthScreenMode.register;
                  final textScale =
                      MediaQuery.textScalerOf(context).scale(14) / 14;
                  final compact =
                      isRegister &&
                      (constraints.maxHeight < 650 || textScale > 1.15);
                  final hideIntroduction =
                      isRegister &&
                      (keyboardOpen ||
                          compact ||
                          _showValidationErrors ||
                          _errorMessage != null);
                  final faceHeight = !isRegister
                      ? 148.0
                      : hideIntroduction
                      ? 0.0
                      : 100.0;
                  final shortHeading =
                      keyboardOpen ||
                      _showValidationErrors ||
                      _errorMessage != null;
                  final padding = isRegister
                      ? EdgeInsets.fromLTRB(
                          24,
                          keyboardOpen ? 2 : 10,
                          24,
                          keyboardOpen ? 2 : 10,
                        )
                      : const EdgeInsets.fromLTRB(24, 18, 24, 28);
                  final minimumContentHeight = math.max(
                    0.0,
                    constraints.maxHeight - padding.vertical,
                  );

                  final content = Column(
                    mainAxisSize: MainAxisSize.min,
                    crossAxisAlignment: CrossAxisAlignment.stretch,
                    children: [
                      if (!isRegister)
                        ListenableBuilder(
                          listenable: _secureFieldFocus,
                          builder: (context, _) => _AuthFace(
                            animation: _faceController,
                            palette: palette,
                            reduceMotion: _reduceMotion,
                            eyesClosed: _isEnteringPassword,
                          ),
                        ),
                      if (isRegister)
                        SizedBox(
                          height: faceHeight,
                          child: FittedBox(
                            fit: BoxFit.contain,
                            child: SizedBox(
                              width: 345,
                              child: ListenableBuilder(
                                listenable: _secureFieldFocus,
                                builder: (context, _) => _AuthFace(
                                  animation: _faceController,
                                  palette: palette,
                                  reduceMotion: _reduceMotion,
                                  eyesClosed: _isEnteringPassword,
                                ),
                              ),
                            ),
                          ),
                        ),
                      if (faceHeight > 0) SizedBox(height: isRegister ? 6 : 14),
                      Text(
                        _mode == AuthScreenMode.register
                            ? shortHeading
                                  ? 'Create account'
                                  : 'A joyful place starts with you'
                            : 'Welcome back, grown-up',
                        textAlign: TextAlign.center,
                        style: theme.textTheme.headlineSmall?.copyWith(
                          color: palette.foreground,
                          fontSize: isRegister
                              ? shortHeading
                                    ? 23
                                    : compact
                                    ? 26
                                    : 30
                              : 33,
                          fontWeight: FontWeight.w800,
                          height: 1.04,
                          letterSpacing: -1,
                        ),
                      ),
                      if (!hideIntroduction) ...[
                        SizedBox(height: isRegister ? 6 : 10),
                        Text(
                          _mode == AuthScreenMode.register
                              ? 'Create your parent account, then make a fun profile for each little explorer.'
                              : 'Sign in to manage your family\'s Kingdom Kids profiles.',
                          textAlign: TextAlign.center,
                          style: theme.textTheme.bodyMedium?.copyWith(
                            color: palette.muted,
                            fontSize: 14.5,
                            fontWeight: FontWeight.w500,
                            height: 1.42,
                          ),
                        ),
                      ],
                      SizedBox(
                        height: isRegister ? (keyboardOpen ? 2 : 14) : 24,
                      ),
                      AutofillGroup(
                        child: Form(
                          key: _formKey,
                          child: Column(
                            crossAxisAlignment: CrossAxisAlignment.stretch,
                            children: [
                              _AuthTextField(
                                key: const Key('auth-email-field'),
                                controller: _emailController,
                                palette: palette,
                                compact: isRegister,
                                label: 'Email address',
                                hint: 'you@example.com',
                                keyboardType: TextInputType.emailAddress,
                                autofillHints: const [AutofillHints.email],
                                validator: _validateEmail,
                              ),
                              SizedBox(
                                height: isRegister
                                    ? (keyboardOpen ? 4 : 10)
                                    : 13,
                              ),
                              _AuthTextField(
                                key: const Key('auth-password-field'),
                                controller: _passwordController,
                                focusNode: _passwordFocusNode,
                                palette: palette,
                                compact: isRegister,
                                label: 'Password',
                                hint: _mode == AuthScreenMode.register
                                    ? 'At least 8 characters'
                                    : 'Your password',
                                obscureText: !_showPassword,
                                autofillHints: [
                                  _mode == AuthScreenMode.register
                                      ? AutofillHints.newPassword
                                      : AutofillHints.password,
                                ],
                                suffixIcon: IconButton(
                                  tooltip: _showPassword
                                      ? 'Hide password'
                                      : 'Show password',
                                  onPressed: () => setState(
                                    () => _showPassword = !_showPassword,
                                  ),
                                  icon: Icon(
                                    _showPassword
                                        ? Icons.visibility_off_rounded
                                        : Icons.visibility_rounded,
                                  ),
                                ),
                                validator: _validatePassword,
                                onFieldSubmitted: (_) {
                                  if (_mode == AuthScreenMode.login) {
                                    _submit();
                                  }
                                },
                              ),
                              if (_mode == AuthScreenMode.login) ...[
                                const SizedBox(height: 4),
                                Align(
                                  alignment: Alignment.centerRight,
                                  child: TextButton(
                                    key: const Key('auth-forgot-password'),
                                    onPressed: _isSubmitting
                                        ? null
                                        : () => Navigator.of(
                                            context,
                                          ).pushNamed(AppRoutes.passwordReset),
                                    style: TextButton.styleFrom(
                                      foregroundColor: palette.foreground,
                                      minimumSize: const Size(48, 44),
                                    ),
                                    child: const Text(
                                      'Forgot password?',
                                      style: TextStyle(
                                        fontWeight: FontWeight.w700,
                                      ),
                                    ),
                                  ),
                                ),
                              ],
                              if (_mode == AuthScreenMode.register) ...[
                                SizedBox(height: keyboardOpen ? 4 : 10),
                                _AuthTextField(
                                  key: const Key('auth-confirm-password-field'),
                                  controller: _confirmPasswordController,
                                  focusNode: _confirmPasswordFocusNode,
                                  palette: palette,
                                  compact: true,
                                  label: 'Confirm password',
                                  hint: 'One more time',
                                  obscureText: !_showConfirmPassword,
                                  autofillHints: const [
                                    AutofillHints.newPassword,
                                  ],
                                  suffixIcon: IconButton(
                                    tooltip: _showConfirmPassword
                                        ? 'Hide password'
                                        : 'Show password',
                                    onPressed: () => setState(
                                      () => _showConfirmPassword =
                                          !_showConfirmPassword,
                                    ),
                                    icon: Icon(
                                      _showConfirmPassword
                                          ? Icons.visibility_off_rounded
                                          : Icons.visibility_rounded,
                                    ),
                                  ),
                                  validator: (value) {
                                    if (value != _passwordController.text) {
                                      return 'Those passwords do not match.';
                                    }
                                    return null;
                                  },
                                  onFieldSubmitted: (_) => _submit(),
                                ),
                              ],
                            ],
                          ),
                        ),
                      ),
                      if (_errorMessage != null) ...[
                        const SizedBox(height: 15),
                        _ErrorNotice(message: _errorMessage!, palette: palette),
                      ],
                      SizedBox(
                        height: isRegister ? (keyboardOpen ? 6 : 8) : 14,
                      ),
                      FilledButton(
                        key: const Key('auth-submit-button'),
                        style: FilledButton.styleFrom(
                          backgroundColor: palette.action,
                          foregroundColor: palette.onAction,
                          disabledBackgroundColor: palette.action.withValues(
                            alpha: 0.48,
                          ),
                          disabledForegroundColor: palette.onAction.withValues(
                            alpha: 0.7,
                          ),
                          minimumSize: Size.fromHeight(isRegister ? 50 : 57),
                          shape: const StadiumBorder(),
                          textStyle: const TextStyle(
                            fontSize: 16,
                            fontWeight: FontWeight.w800,
                            letterSpacing: 0.1,
                          ),
                        ),
                        onPressed: _isSubmitting ? null : _submit,
                        child: _isSubmitting
                            ? SizedBox.square(
                                dimension: 22,
                                child: CircularProgressIndicator(
                                  strokeWidth: 2.4,
                                  color: palette.onAction,
                                ),
                              )
                            : Text(
                                _mode == AuthScreenMode.register
                                    ? 'I\'m ready'
                                    : 'Log in',
                              ),
                      ),
                      SizedBox(
                        height: isRegister ? (keyboardOpen ? 0 : 2) : 12,
                      ),
                      _AuthPrivacyPolicyLink(
                        mode: _mode,
                        palette: palette,
                        onPressed: _isSubmitting ? null : _openPrivacyPolicy,
                      ),
                      SizedBox(height: isRegister ? 0 : 10),
                      _InlineModeLink(
                        mode: _mode,
                        palette: palette,
                        onChanged: _changeMode,
                      ),
                      SizedBox(height: isRegister ? 4 : 11),
                      _ParentSafetyNote(palette: palette, compact: isRegister),
                    ],
                  );

                  // Preserve the user's accessible type size when a very large
                  // setting cannot physically fit the complete form at once.
                  if (isRegister && textScale < 1.7) {
                    return Padding(
                      key: const Key('auth-register-viewport'),
                      padding: padding,
                      child: Center(
                        child: ConstrainedBox(
                          constraints: const BoxConstraints(maxWidth: 460),
                          child: content,
                        ),
                      ),
                    );
                  }
                  return SingleChildScrollView(
                    keyboardDismissBehavior:
                        ScrollViewKeyboardDismissBehavior.onDrag,
                    padding: padding,
                    child: Center(
                      child: ConstrainedBox(
                        constraints: BoxConstraints(
                          maxWidth: 460,
                          minHeight: minimumContentHeight,
                        ),
                        child: IntrinsicHeight(child: content),
                      ),
                    ),
                  );
                },
              ),
            ),
          ],
        ),
      ),
    );
  }

  String? _validateEmail(String? value) {
    final email = value?.trim() ?? '';
    if (email.isEmpty) {
      return 'Please enter your email address.';
    }
    if (!RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(email)) {
      return 'Enter a valid email address.';
    }
    return null;
  }

  String? _validatePassword(String? value) {
    if (value == null || value.isEmpty) {
      return 'Please enter your password.';
    }
    if (_mode == AuthScreenMode.register && value.length < 8) {
      return 'Use at least 8 characters.';
    }
    return null;
  }
}

class _AuthFace extends StatelessWidget {
  const _AuthFace({
    required this.animation,
    required this.palette,
    required this.reduceMotion,
    required this.eyesClosed,
  });

  final Animation<double> animation;
  final _AuthPalette palette;
  final bool reduceMotion;
  final bool eyesClosed;

  @override
  Widget build(BuildContext context) {
    return Semantics(
      key: const Key('auth-face'),
      container: true,
      image: true,
      label: 'Playful Kingdom Kids face',
      value: eyesClosed
          ? 'Eyes closed while password is being entered'
          : 'Eyes open',
      child: SizedBox(
        height: 148,
        child: TweenAnimationBuilder<double>(
          key: const Key('auth-eye-closure-animation'),
          tween: Tween<double>(begin: 0, end: eyesClosed ? 1 : 0),
          duration: reduceMotion
              ? Duration.zero
              : Duration(milliseconds: eyesClosed ? 190 : 260),
          curve: eyesClosed ? Curves.easeInCubic : Curves.easeOutCubic,
          builder: (context, eyeClosure, _) {
            return AnimatedBuilder(
              animation: animation,
              builder: (context, _) {
                final progress = reduceMotion ? 0.5 : animation.value;
                final gazeX = (progress - 0.5) * 0.76;
                final gazeY = reduceMotion
                    ? 0.16
                    : 0.12 + math.sin(progress * math.pi) * 0.11;
                final bob = reduceMotion
                    ? 0.0
                    : -3.5 * math.sin(progress * math.pi);

                return Transform.translate(
                  offset: Offset(0, bob),
                  child: Stack(
                    alignment: Alignment.topCenter,
                    clipBehavior: Clip.none,
                    children: [
                      Positioned(
                        top: 0,
                        child: CustomPaint(
                          size: const Size(54, 28),
                          painter: _SquigglePainter(color: palette.foreground),
                        ),
                      ),
                      Positioned(
                        top: 34,
                        child: Row(
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            Transform.rotate(
                              angle: -0.045,
                              child: _AuthEye(
                                palette: palette,
                                gazeX: gazeX,
                                gazeY: gazeY,
                                closure: eyeClosure,
                              ),
                            ),
                            const SizedBox(width: 8),
                            Transform.rotate(
                              angle: 0.045,
                              child: _AuthEye(
                                palette: palette,
                                gazeX: gazeX,
                                gazeY: gazeY,
                                closure: eyeClosure,
                              ),
                            ),
                          ],
                        ),
                      ),
                      Positioned(
                        top: 98,
                        child: CustomPaint(
                          size: const Size(29, 22),
                          painter: _NosePainter(color: palette.nose),
                        ),
                      ),
                      Positioned(
                        top: 117,
                        left: 74,
                        child: _Cheek(color: palette.nose),
                      ),
                      Positioned(

class _ClosedEyePainter extends CustomPainter {
  const _ClosedEyePainter({required this.color});

  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeWidth = 5;
    final path = Path()
      ..moveTo(3, size.height * 0.32)
      ..quadraticBezierTo(
        size.width / 2,
        size.height * 0.9,
        size.width - 3,
        size.height * 0.32,
      );
    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(_ClosedEyePainter oldDelegate) =>
      oldDelegate.color != color;
}

class _SquigglePainter extends CustomPainter {
  const _SquigglePainter({required this.color});

  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round
      ..strokeWidth = 3.2;
    final path = Path()
      ..moveTo(2, size.height - 3)
      ..cubicTo(7, 4, 14, 3, size.width / 2, size.height - 4)
      ..cubicTo(size.width - 14, 3, size.width - 7, 4, size.width - 2, 12);
    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(_SquigglePainter oldDelegate) =>
      oldDelegate.color != color;
}

class _NosePainter extends CustomPainter {
  const _NosePainter({required this.color});

  final Color color;

  @override
  void paint(Canvas canvas, Size size) {
    final path = Path()
      ..moveTo(size.width / 2, size.height)
      ..lineTo(0, 0)
      ..quadraticBezierTo(size.width / 2, 5, size.width, 0)
      ..close();
    canvas.drawPath(path, Paint()..color = color);
  }

  @override
  bool shouldRepaint(_NosePainter oldDelegate) => oldDelegate.color != color;
}

class _AuthTextField extends StatelessWidget {
  const _AuthTextField({
    super.key,
    required this.controller,
    this.focusNode,
    this.compact = false,
    required this.palette,
    required this.label,
    required this.hint,
    this.keyboardType,
    this.autofillHints,
    this.obscureText = false,
    this.suffixIcon,
    this.validator,
    this.onFieldSubmitted,
  });

  final TextEditingController controller;
  final FocusNode? focusNode;
  final bool compact;
  final _AuthPalette palette;
  final String label;
  final String hint;
  final TextInputType? keyboardType;
  final Iterable<String>? autofillHints;
  final bool obscureText;
  final Widget? suffixIcon;
  final FormFieldValidator<String>? validator;
  final ValueChanged<String>? onFieldSubmitted;

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: controller,
      focusNode: focusNode,
      keyboardType: keyboardType,
      autofillHints: autofillHints,
      obscureText: obscureText,
      enableSuggestions: !obscureText,
      autocorrect: !obscureText,
      textInputAction: onFieldSubmitted == null
          ? TextInputAction.next
          : TextInputAction.done,
      validator: validator,
      onFieldSubmitted: onFieldSubmitted,
      cursorColor: palette.focus,
      style: TextStyle(
        color: palette.fieldText,
        fontSize: 15,
        fontWeight: FontWeight.w600,
      ),
      decoration: InputDecoration(
        labelText: label,
        hintText: hint,
        suffixIcon: suffixIcon,
        suffixIconColor: palette.fieldHint,
        filled: true,
        fillColor: palette.fieldFill,
        labelStyle: TextStyle(
          color: palette.fieldHint,
          fontWeight: FontWeight.w600,
        ),
        floatingLabelStyle: TextStyle(
          color: palette.focus,
          fontWeight: FontWeight.w800,
        ),
        hintStyle: TextStyle(
          color: palette.fieldHint,
          fontWeight: FontWeight.w500,
        ),
        errorStyle: TextStyle(
          color: palette.errorText,
          fontWeight: FontWeight.w700,
        ),
        isDense: compact,
        contentPadding: EdgeInsets.symmetric(
          horizontal: 20,
          vertical: compact ? 12 : 16,
        ),
        border: _border(),
        enabledBorder: _border(),
        focusedBorder: _border(color: palette.focus, width: 2),
        errorBorder: _border(color: palette.errorText, width: 1.5),
        focusedErrorBorder: _border(color: palette.errorText, width: 2),
      ),
    );
  }

  OutlineInputBorder _border({
    Color color = Colors.transparent,
    double width = 1,
  }) {
    return OutlineInputBorder(
      borderRadius: BorderRadius.circular(28),
      borderSide: BorderSide(color: color, width: width),
    );
  }
}
